  <script>
    const form = document.getElementById("bookingForm");
    const dateInput = document.getElementById("bookingDate");
    const timeInput = document.getElementById("bookingTime");
    const statusBox = document.getElementById("queueStatus");
    const submitBtn = document.getElementById("submitBtn");
    const lineLoginStatus = document.getElementById("lineLoginStatus");

    // LIFF ID ของระบบจองคิว CPAP
    const LIFF_ID = "2012251258-lkZs5XFV";
    let lineIdToken = "";
    let liffReady = false;

    let datePicker = null;
    let checkSequence = 0;
    let checkTimer = null;
    let isChecking = false;
    let isSubmitting = false;
    let lastCheckKey = "";
    let lastCheckAvailable = null;

    // -------------------------------------------------
    // เริ่มต้น LIFF และเตรียม ID Token สำหรับยืนยันตัวตน
    // ใน LINE Developers Console ต้องเปิด scope: openid
    // -------------------------------------------------
    async function initializeLiff() {
      try {
        await liff.init({ liffId: LIFF_ID });

        if (!liff.isLoggedIn() && !liff.isInClient()) {
          lineLoginStatus.textContent = "กำลังเปิดหน้าเข้าสู่ระบบ LINE...";
          liff.login({ redirectUri: window.location.href });
          return;
        }

        lineIdToken = liff.getIDToken() || "";
        if (!lineIdToken) {
          throw new Error("ไม่พบ ID Token กรุณาตรวจสอบว่าเปิด scope openid ใน LIFF แล้ว");
        }

        liffReady = true;
        submitBtn.disabled = false;
        lineLoginStatus.textContent = "เชื่อมต่อ LINE สำเร็จ สามารถจองคิวได้";
      } catch (error) {
        console.error("LIFF initialization error:", error);
        liffReady = false;
        submitBtn.disabled = true;
        lineLoginStatus.textContent = "เชื่อมต่อ LINE ไม่สำเร็จ กรุณาเปิดแบบฟอร์มผ่าน LIFF และตรวจสอบการตั้งค่า openid";
        Swal.fire({
          icon: "error",
          title: "เชื่อมต่อ LINE ไม่สำเร็จ",
          text: "กรุณาเปิดแบบฟอร์มจาก LIFF ใน LINE และตรวจสอบว่า LIFF เปิด scope openid แล้ว",
          confirmButtonText: "ตกลง"
        });
      }
    }

    initializeLiff();

    // -------------------------------------------------
    // แสดงวันที่เป็น พ.ศ. เช่น 6 ตุลาคม 2569
    // -------------------------------------------------

    function formatThaiDate(date) {
      const months = [
        "มกราคม", "กุมภาพันธ์", "มีนาคม", "เมษายน",
        "พฤษภาคม", "มิถุนายน", "กรกฎาคม", "สิงหาคม",
        "กันยายน", "ตุลาคม", "พฤศจิกายน", "ธันวาคม"
      ];

      return (
        date.getDate() + " " +
        months[date.getMonth()] + " " +
        (date.getFullYear() + 543)
      );
    }

    // -------------------------------------------------
    // แสดงสถานะโดยไม่เปิดเผยจำนวนคิว
    // -------------------------------------------------

    function showStatus(type, message) {
      statusBox.className = "queue-status " + type;
      statusBox.textContent = message;
    }

    function clearStatus() {
      statusBox.className = "queue-status";
      statusBox.textContent = "";
    }

    function selectedKey() {
      return dateInput.value + "|" + timeInput.value;
    }

    // -------------------------------------------------
    // เริ่มต้นปฏิทิน
    // -------------------------------------------------

    document.addEventListener("DOMContentLoaded", function () {
      datePicker = flatpickr(dateInput, {
        locale: "th",
        dateFormat: "Y-m-d",
        altInput: true,
        altFormat: "j F Y",
        minDate: "today",
        disable: [
          function (date) {
            return date.getDay() === 0 || date.getDay() === 6;
          }
        ],

        onReady: function (selectedDates, dateStr, instance) {
          if (instance.altInput && selectedDates.length) {
            instance.altInput.value = formatThaiDate(selectedDates[0]);
          }

          if (instance.altInput) {
            instance.altInput.placeholder = "กรุณาเลือกวันที่จากปฏิทิน";
            instance.altInput.readOnly = true;
            instance.altInput.classList.add("form-control");
          }
        },

        onChange: function (selectedDates, dateStr, instance) {
          if (instance.altInput && selectedDates.length) {
            instance.altInput.value = formatThaiDate(selectedDates[0]);
          }

          lastCheckKey = "";
          lastCheckAvailable = null;
          checkQueueAvailability();
        }
      });
    });

    // -------------------------------------------------
    // ตรวจคิวเมื่อเลือกวันและเวลา
    // หน่วงเพียงเล็กน้อยเพื่อรวมการเปลี่ยนแปลงที่ติดกัน
    // -------------------------------------------------

    function checkQueueAvailability() {
      clearTimeout(checkTimer);

      const bookingDate = dateInput.value;
      const bookingTime = timeInput.value;

      if (!bookingDate || !bookingTime) {
        checkSequence++;
        isChecking = false;
        lastCheckKey = "";
        lastCheckAvailable = null;
        clearStatus();
        return;
      }

      const key = selectedKey();

      // ถ้าคู่วันและเวลานี้ตรวจเสร็จแล้ว ใช้ผลเดิมได้
      if (key === lastCheckKey && lastCheckAvailable !== null) {
        showStatus(
          lastCheckAvailable ? "available" : "full",
          lastCheckAvailable
            ? "✓ ช่วงเวลานี้ยังว่าง สามารถจองได้"
            : "✕ ช่วงเวลานี้คิวเต็มแล้ว กรุณาเลือกเวลาอื่น"
        );
        return;
      }

      const sequence = ++checkSequence;
      isChecking = true;

      showStatus("checking", "กำลังตรวจสอบคิว...");

      checkTimer = setTimeout(function () {
        google.script.run
          .withSuccessHandler(function (result) {
            // ทิ้งผลลัพธ์เก่าถ้าผู้ใช้เปลี่ยนวันหรือเวลาไปแล้ว
            if (sequence !== checkSequence || key !== selectedKey()) {
              return;
            }

            isChecking = false;
            lastCheckKey = key;

            if (!result || !result.success) {
              lastCheckAvailable = null;
              showStatus(
                "error",
                result && result.message
                  ? result.message
                  : "ตรวจสอบคิวไม่สำเร็จ กรุณาลองใหม่"
              );
              return;
            }

            lastCheckAvailable = result.available === true;

            showStatus(
              lastCheckAvailable ? "available" : "full",
              lastCheckAvailable
                ? "✓ ช่วงเวลานี้ยังว่าง สามารถจองได้"
                : "✕ ช่วงเวลานี้คิวเต็มแล้ว กรุณาเลือกเวลาอื่น"
            );
          })
          .withFailureHandler(function (error) {
            if (sequence !== checkSequence) {
              return;
            }

            isChecking = false;
            lastCheckAvailable = null;

            console.error(error);

            showStatus(
              "error",
              "ตรวจสอบคิวไม่สำเร็จ กรุณาลองอีกครั้ง"
            );
          })
          .checkQueue(bookingDate, bookingTime);
      }, 100);
    }

    timeInput.addEventListener("change", function () {
      lastCheckKey = "";
      lastCheckAvailable = null;
      checkQueueAvailability();
    });

    // -------------------------------------------------
    // รับเฉพาะตัวเลขในช่องเบอร์โทร
    // -------------------------------------------------

    document.getElementById("phone").addEventListener("input", function () {
      this.value = this.value.replace(/\D/g, "").slice(0, 10);
    });

    // -------------------------------------------------
    // ส่งแบบฟอร์ม
    // -------------------------------------------------

    form.addEventListener("submit", function (event) {
      event.preventDefault();

      if (isSubmitting) {
        return;
      }

      if (!liffReady || !lineIdToken) {
        Swal.fire({
          icon: "warning",
          title: "ยังเชื่อมต่อ LINE ไม่สำเร็จ",
          text: "กรุณาเปิดแบบฟอร์มผ่าน LIFF แล้วลองใหม่อีกครั้ง",
          confirmButtonText: "ตกลง"
        });
        return;
      }

      const firstName = document.getElementById("firstName").value.trim();
      const lastName = document.getElementById("lastName").value.trim();
      const phone = document.getElementById("phone").value.trim();
      const cpapBrand = document.getElementById("cpapBrand").value.trim();
      const bookingDate = dateInput.value;
      const bookingTime = timeInput.value;
      const note = document.getElementById("note").value.trim();

      if (!firstName || !lastName || !phone || !cpapBrand || !bookingDate || !bookingTime) {
        Swal.fire({
          icon: "warning",
          title: "กรุณากรอกข้อมูลให้ครบ",
          confirmButtonText: "ตกลง"
        });
        return;
      }

      if (!/^0\d{9}$/.test(phone)) {
        Swal.fire({
          icon: "warning",
          title: "เบอร์โทรศัพท์ไม่ถูกต้อง",
          text: "กรุณากรอกเบอร์โทรศัพท์ 10 หลัก",
          confirmButtonText: "ตกลง"
        });
        return;
      }

      // ถ้ากำลังตรวจคิวอยู่ อย่าเพิ่งส่งข้อมูล
      if (isChecking) {
        Swal.fire({
          icon: "info",
          title: "กำลังตรวจสอบคิว",
          text: "กรุณารอสักครู่แล้วกดยืนยันอีกครั้ง",
          confirmButtonText: "ตกลง"
        });
        return;
      }

      // ถ้าผลตรวจล่าสุดบอกว่าเต็ม ให้หยุดทันที
      if (selectedKey() === lastCheckKey && lastCheckAvailable === false) {
        Swal.fire({
          icon: "warning",
          title: "คิวเต็มแล้ว",
          text: "ช่วงเวลานี้มีผู้จองครบแล้ว กรุณาเลือกเวลาอื่น",
          confirmButtonText: "ตกลง"
        });
        return;
      }

      isSubmitting = true;
      submitBtn.disabled = true;
      submitBtn.textContent = "กำลังบันทึกข้อมูล...";

      const formData = {
        firstName: firstName,
        lastName: lastName,
        phone: phone,
        cpapBrand: cpapBrand,
        bookingDate: bookingDate,
        bookingTime: bookingTime,
        note: note,
        idToken: lineIdToken
      };

      google.script.run
        .withSuccessHandler(function (result) {
          isSubmitting = false;
          submitBtn.disabled = !liffReady;
          submitBtn.textContent = "ยืนยันการจองคิว";

          if (!result || !result.success) {
            lastCheckKey = "";
            lastCheckAvailable = null;

            Swal.fire({
              icon: "warning",
              title: "ไม่สามารถจองคิวได้",
              text: result && result.message
                ? result.message
                : "กรุณาตรวจสอบวันและเวลาอีกครั้ง",
              confirmButtonText: "ตกลง"
            }).then(function () {
              checkQueueAvailability();
            });

            return;
          }

          const parts = bookingDate.split("-").map(Number);
          const dateObject = new Date(parts[0], parts[1] - 1, parts[2]);

          Swal.fire({
            icon: "success",
            title: "จองคิวสำเร็จ",
            html:
              "ระบบบันทึกข้อมูลการจองเรียบร้อยแล้ว<br><br>" +
              "<b>วันที่:</b> " + formatThaiDate(dateObject) + "<br>" +
              "<b>เวลา:</b> " + bookingTime + " น.",
            confirmButtonText: "ตกลง"
          }).then(function () {
            form.reset();

            if (datePicker) {
              datePicker.clear();
            }

            lastCheckKey = "";
            lastCheckAvailable = null;
            isChecking = false;
            clearStatus();
          });
        })
        .withFailureHandler(function (error) {
          console.error(error);

          isSubmitting = false;
          submitBtn.disabled = !liffReady;
          submitBtn.textContent = "ยืนยันการจองคิว";

          Swal.fire({
            icon: "error",
            title: "เกิดข้อผิดพลาด",
            text: "ไม่สามารถบันทึกข้อมูลได้ กรุณาลองใหม่อีกครั้ง",
            confirmButtonText: "ตกลง"
          });
        })
        .submitBooking(formData);
    });
  </script>
